Map<String, int> get languagesUsage {
  final String? usage = _sharedPreferences.getString(_TAG_LANGUAGES_USAGE);
  if (usage == null || usage.isEmpty) {
    return <String, int>{};
  }
  return Map<String, int>.from(jsonDecode(usage));
}
