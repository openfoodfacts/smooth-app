static FutureOr<void> onUpgrade(
  final Database db,
  final int oldVersion,
  final int newVersion,
) async {
  if (oldVersion < 5) {
    await db.execute(
      'create table $TABLE('
      // cf. https://www.sqlite.org/lang_conflict.html
      '$COLUMN_BARCODE TEXT PRIMARY KEY on conflict replace'
      ',$COLUMN_LAST_ACCESS INT NOT NULL'
      ')',
    );
  }
}
